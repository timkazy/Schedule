  const handleCellClick = (tableId, colIndex, cellIndex, event) => {
    setActiveTableId(tableId);

    setSelectedCells((prev) => {
      const sameTableCells = prev.filter((c) => c.tableId === tableId);
      const alreadySelected = sameTableCells.some(
        (c) => c.columnIndex === colIndex && c.cellIndex === cellIndex
      );

      // Ctrl — добавляем / убираем выделение
      if (event.ctrlKey || event.metaKey) {
        if (alreadySelected) {
          return prev.filter(
            (c) =>
              !(
                c.tableId === tableId &&
                c.columnIndex === colIndex &&
                c.cellIndex === cellIndex
              )
          );
        } else {
          return [...prev, { tableId, columnIndex: colIndex, cellIndex }];
        }
      }

      // Shift — диапазонное выделение
      if (event.shiftKey && prev.length > 0) {
        const last = prev[prev.length - 1];
        if (last.tableId === tableId) {
          const startCol = Math.min(last.columnIndex, colIndex);
          const endCol = Math.max(last.columnIndex, colIndex);
          const startRow = Math.min(last.cellIndex, cellIndex);
          const endRow = Math.max(last.cellIndex, cellIndex);
          const newSelection = [];
          for (let c = startCol; c <= endCol; c++) {
            for (let r = startRow; r <= endRow; r++) {
              newSelection.push({ tableId, columnIndex: c, cellIndex: r });
            }
          }
          return newSelection;
        }
      }

      // Обычный клик — выделяем только одну
      return [{ tableId, columnIndex: colIndex, cellIndex }];
    });
  };
